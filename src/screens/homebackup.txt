          dispatch(fetchDiscountBanners(cat.id)).unwrap(),

      await dispatch(fetchUnderDeals()).unwrap();
      await dispatch(fetchBrands()).unwrap();

    } finally {
      setRefreshing(false);
    }

        status: err.status,

          prev.map((product) =>
            product.product_id === item.product_id

    if (lastPart.includes('India -')) {
      return lastPart.split('-')[1]?.trim() || 'N/A';

    return lastPart.match(/^\d{6}$/) ? lastPart : 'N/A';

  const handleVideoPress = () => {

    navigation.navigate('SearchScreen');
  };

      <TouchableOpacity
        onPress={() => {

              brandId: item.reference,
              userId: customerId,

              deal: {
                offer: item.reference,
                products: [],

              categoryId: item.category_id,
              offer: item.reference,
              userId: customerId,

          width: '100%',
          height: '100%',

          source={{ uri: item.banner_image }}
          style={{ width: '100%', height: '100%', resizeMode: 'cover' }}

  return (
    <SafeAreaView style={{ flex: 1, backgroundColor: '#fff' }}>

      <ScrollView
        style={{ flex: 1, backgroundColor: '#fff' }}
        refreshControl={

            refreshing={refreshing}
            onRefresh={onRefresh}
            colors={['#5E61EB']}
            tintColor="#5E61EB"
          />
        }
      >

                    end={{ x: 0.5, y: 1 }}

                source={{ uri: promoVideos[0].video }}
                style={styles.video}
                resizeMode="cover"
                repeat={true}
                paused={isVideoPaused}
                onError={(error) => {
                  console.error('Video playback error:', JSON.stringify(error, null, 2));
                }}
                onLoad={() => console.log('Video loaded successfully:', promoVideos[0].video)}
                onLoadStart={() => console.log('Video loading started:', promoVideos[0].video)}
              />
            ) : (
              <View style={styles.errorContainer}>
                <Text style={styles.errorText}>No video available</Text>
              </View>
            )}
            <TouchableOpacity
              style={styles.playPauseButton}
              onPress={() => setIsVideoPaused(!isVideoPaused)}
              accessibilityLabel={isVideoPaused ? 'Play video' : 'Pause video'}
            >
              <Ionicons
                name={isVideoPaused ? 'play' : 'pause'}
                size={24}
                color="#fff"
              />
            </TouchableOpacity>
            <TouchableOpacity
              style={styles.closeButton}
              onPress={() => setShowVideo(false)}
              accessibilityLabel="Close video"
            >
              <Ionicons name="close" size={20} color="#fff" />
            </TouchableOpacity>
            <TouchableOpacity
              style={styles.fullScreenButton}
              onPress={handleVideoPress}
              accessibilityLabel="View video in full screen"
            >
              <Ionicons name="expand" size={20} color="#fff" />
            </TouchableOpacity>
          </TouchableOpacity>
        </View>
      )}
    </SafeAreaView>
  );

  onClose,
  onExplore,

  imageSource,
  buttonText,
  isLoading = false, // New prop to control loading state

  return (
    <Modal visible={visible} transparent animationType="slide" onRequestClose={onClose}>

            source={imageSource}

            resizeMode="contain"
