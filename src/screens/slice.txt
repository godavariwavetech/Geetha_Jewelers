import { createSlice, createAsyncThunk } from '@reduxjs/toolkit';
import { endpoints } from '../../config/config';
import api from '../../utils/api';

export const fetchCategories = createAsyncThunk(
  'category/fetchCategories',
  async (_, { fulfillWithValue, rejectWithValue }) => {
    try {
      const res = await api.get(endpoints.GET_CATEGORIES);
      if (res.data?.data) {
        const limitedCategories = res.data.data.slice(0, 4);
        return fulfillWithValue(limitedCategories);
      } else {
        return rejectWithValue('No category data found');
      }
    } catch (err) {
      console.log(err);
      return rejectWithValue(
        err.response?.data?.message || 'Failed to fetch categories'
      );
    }
  }
);

export const fetchSubcategories = createAsyncThunk(
  'category/fetchSubcategories',
  async (categoryId, { fulfillWithValue, rejectWithValue }) => {
    try {
      const res = await api.post(endpoints.GET_SUBCATEGORY, {
        category_id: categoryId,
      });
      if (res.data?.data) {
        return fulfillWithValue(res.data.data);
      } else {
        return rejectWithValue('No subcategory data found');
      }
    } catch (err) {
      console.log(err);
      return rejectWithValue(
        err.response?.data?.message || 'Failed to fetch subcategories'
      );
    }
  }
);

export const fetchBanners = createAsyncThunk(
  'category/fetchBanners',
  async (categoryId, { fulfillWithValue, rejectWithValue }) => {
    try {
      const res = await api.post(endpoints.GET_BANNERS, {
        category_id: categoryId,
      });
      if (res.data?.data) {
        return fulfillWithValue(res.data.data);
      } else {
        return rejectWithValue('No banner data found');
      }
    } catch (err) {
      console.log(err);
      return rejectWithValue(
        err.response?.data?.message || 'Failed to fetch banners'
      );
    }
  }
);

export const fetchBrands = createAsyncThunk(
  'category/fetchBrands',
  async (_, { fulfillWithValue, rejectWithValue }) => {
    try {
      const res = await api.get(endpoints.GET_BRANDS);
      if (res.data?.data) {
        return fulfillWithValue(res.data.data);
      } else {
        return rejectWithValue('No brands data found');
      }
    } catch (err) {
      console.log(err);
      return rejectWithValue(
        err.response?.data?.message || 'Failed to fetch brands'
      );
    }
  }
);

export const fetchDiscountBanners = createAsyncThunk(
  'category/fetchDiscountBanners',
  async (categoryId, { fulfillWithValue, rejectWithValue }) => {
    try {
      const res = await api.post(endpoints.GET_OFFERBANNER, {
        category_id: categoryId,
      });
      if (res.data?.data) {
        return fulfillWithValue(res.data.data);
      } else {
        return rejectWithValue('No discount banners found');
      }
    } catch (err) {
      console.log(err);
      return rejectWithValue(
        err.response?.data?.message || 'Failed to fetch discount banners'
      );
    }
  }
);

const categorySlice = createSlice({
  name: 'category',
  initialState: {
    categories: [],
    subcategories: [],
    loading: false,
    error: null,
    subLoading: false,
    subError: null,
    banners: [],
    bannersLoading: false,
    bannersError: null,
    brands: [],
    brandsLoading: false,
    brandsError: null,
    discountBanners: [],
    discountBannersLoading: false,
    discountBannersError: null,
  },
  reducers: {},
  extraReducers: (builder) => {
    builder
      .addCase(fetchCategories.pending, (state) => {
        state.loading = true;
        state.error = null;
      })
      .addCase(fetchCategories.fulfilled, (state, action) => {
        state.loading = false;
        state.categories = action.payload;
      })
      .addCase(fetchCategories.rejected, (state, action) => {
        state.loading = false;
        state.error = action.payload;
      })

      .addCase(fetchSubcategories.pending, (state) => {
        state.subLoading = true;
        state.subError = null;
      })
      .addCase(fetchSubcategories.fulfilled, (state, action) => {
        state.subLoading = false;
        state.subcategories = action.payload;
      })
      .addCase(fetchSubcategories.rejected, (state, action) => {
        state.subLoading = false;
        state.subError = action.payload;
      })

      .addCase(fetchBanners.pending, (state) => {
        state.bannersLoading = true;
        state.bannersError = null;
      })
      .addCase(fetchBanners.fulfilled, (state, action) => {
        state.bannersLoading = false;
        state.banners = action.payload;
      })
      .addCase(fetchBanners.rejected, (state, action) => {
        state.bannersLoading = false;
        state.bannersError = action.payload;
      })

      .addCase(fetchBrands.pending, (state) => {
        state.brandsLoading = true;
        state.brandsError = null;
      })
      .addCase(fetchBrands.fulfilled, (state, action) => {
        state.brandsLoading = false;
        state.brands = action.payload;
      })
      .addCase(fetchBrands.rejected, (state, action) => {
        state.brandsLoading = false;
        state.brandsError = action.payload;
      })

      .addCase(fetchDiscountBanners.pending, (state) => {
        state.discountBannersLoading = true;
        state.discountBannersError = null;
      })
      .addCase(fetchDiscountBanners.fulfilled, (state, action) => {
        state.discountBannersLoading = false;
        state.discountBanners = action.payload;
      })
      .addCase(fetchDiscountBanners.rejected, (state, action) => {
        state.discountBannersLoading = false;
        state.discountBannersError = action.payload;
      });
  },
});

export default categorySlice.reducer;

import { createSlice } from '@reduxjs/toolkit';

const categorySlice = createSlice({
  name: 'category',
  initialState: {
    productDetails: null,
    recommendedProducts: [],
    subcategories: [],
    productDetailsLoading: false,
    recommendedProductsLoading: false,
    subLoading: false,
    subError: null,
  },
  reducers: {
    updateRecommendedProductWishlist: (state, action) => {
      const { product_id, wishlist_flag, wishlist_id } = action.payload;
      state.recommendedProducts = state.recommendedProducts.map((prod) =>
        prod.product_id === product_id
          ? { ...prod, wishlist_flag, wishlist_id }
          : prod
      );
    },
  },
  extraReducers: (builder) => {
    // Existing fetchProductDetails, fetchRecommendedProducts, fetchSubcategories cases
  },
});

export const { updateRecommendedProductWishlist } = categorySlice.actions;
export default categorySlice.reducer;

const handleRecommendedWishlistToggle = async (item) => {
  const productId = item.product_id;
  if (recommendedWishlistLoading[productId]) return;

  setRecommendedWishlistLoading((prev) => ({ ...prev, [productId]: true }));
  const newWishlistStatus = !item.wishlist_flag;

  // Optimistic update via Redux
  dispatch(
    updateRecommendedProductWishlist({
      product_id: productId,
      wishlist_flag: newWishlistStatus,
      wishlist_id: newWishlistStatus ? item.wishlist_id : null,
    })
  );

  try {
    if (newWishlistStatus) {
      if (!item.varient_id && !item.variant_id) {
        throw new Error('Variant ID is missing');
      }
      const result = await dispatch(
        addToWishlist({
          user_id,
          product_id: productId,
          varient_id: item.varient_id || item.variant_id,
        })
      ).unwrap();
      dispatch(
        updateRecommendedProductWishlist({
          product_id: productId,
          wishlist_flag: true,
          wishlist_id: result.wishlist_id,
        })
      );
    } else {
      if (!item.wishlist_id) throw new Error('Wishlist ID is missing');
      await dispatch(removeFromWishlist({ wishlist_id: item.wishlist_id })).unwrap();
      dispatch(
        updateRecommendedProductWishlist({
          product_id: productId,
          wishlist_flag: false,
          wishlist_id: null,
        })
      );
    }
    // Optional: Refetch to ensure consistency
    dispatch(fetchRecommendedProducts({ user_id, subcategory_id: product.subcategory_id }));
  } catch (error) {
    // Revert optimistic update
    dispatch(
      updateRecommendedProductWishlist({
        product_id: productId,
        wishlist_flag: !newWishlistStatus,
        wishlist_id: item.wishlist_id,
      })
    );
    console.error('Recommended Wishlist error:', error);
    const errorMessage = typeof error === 'string' ? error : error.message || JSON.stringify(error);
    Alert.alert('Error', errorMessage || 'Wishlist action failed');
  } finally {
    setRecommendedWishlistLoading((prev) => ({ ...prev, [productId]: false }));
  }
};

import React, { useState, useEffect } from 'react';
import {
  View,
  Text,
  Image,
  ScrollView,
  TouchableOpacity,
  StyleSheet,

} from '../redux/slices/categorySlice';

const { width } = Dimensions.get('window');

const ProductDetailsScreen = ({ navigation }) => {
  const route = useRoute();
  const dispatch = useDispatch();
  const { user_id, product_id } = route.params;
  const product = useSelector((state) => state.category.productDetails);
  const recommendedProducts = useSelector((state) => state.category.recommendedProducts);
  const subcategories = useSelector((state) => state.category.subcategories);
  const loading = useSelector((state) => state.category.productDetailsLoading);
  const recommendedLoading = useSelector((state) => state.category.recommendedProductsLoading);
  const subLoading = useSelector((state) => state.category.subLoading);
  const subError = useSelector((state) => state.category.subError);
  const wishlistLoading = useSelector((state) => state.wishlist.loading);
  const wishlistError = useSelector((state) => state.wishlist.error);
  const [selectedVariantIndex, setSelectedVariantIndex] = useState(0);
  const [selectedSizeId, setSelectedSizeId] = useState(null);
  const [isWishlisted, setIsWishlisted] = useState(false);
  const [wishlistId, setWishlistId] = useState(null);
  // Local state for recommended products to handle UI updates
  const [localRecommendedProducts, setLocalRecommendedProducts] = useState([]);
  // Track wishlist loading states for recommended products
  const [recommendedWishlistLoading, setRecommendedWishlistLoading] = useState({});
  const insets = useSafeAreaInsets();
  // Sync localRecommendedProducts with Redux recommendedProducts
  useEffect(() => {
    setLocalRecommendedProducts(recommendedProducts);
  }, [recommendedProducts]);
  useEffect(() => {
    dispatch(fetchProductDetails({ user_id, product_id }));
  }, [user_id, product_id, dispatch]);

  useEffect(() => {
    if (product?.subcategory_id) {
      dispatch(fetchRecommendedProducts({ user_id, subcategory_id: product.subcategory_id ,product_id}));
    } else if (product?.category_id) {
      dispatch(fetchSubcategories(product.category_id));
    }
  }, [product, user_id, dispatch]);

  useEffect(() => {
    if (subcategories.length > 0 && !product?.subcategory_id && product?.subcategory_name) {
      const subcategory = subcategories.find(
        (sub) => sub.sub_category_name === product.subcategory_name
      );
      if (subcategory?.id) {
        dispatch(fetchRecommendedProducts({ user_id, subcategory_id: subcategory.id }));
      }
    }
  }, [subcategories, product, user_id, dispatch]);

  useEffect(() => {
    if (product?.variants?.length > 0) {
      const defaultSizeId = product.variants[0].sizes?.[0]?.size_id;
      setSelectedSizeId(defaultSizeId);
      setIsWishlisted(product.variants[0].is_wishlisted || false);
      setWishlistId(product.variants[0].wishlist_id || null);
    }
  }, [product]);

  const handleVariantChange = (index) => {
    setSelectedVariantIndex(index);
    const variant = product.variants[index];
    setSelectedSizeId(variant.sizes[0]?.size_id || null);
    setIsWishlisted(variant.is_wishlisted || false);
    setWishlistId(variant.wishlist_id || null);
  };

  const toggleWishlist = async () => {
    if (wishlistLoading) return;
    const variant = product.variants[selectedVariantIndex];
    const newWishlistStatus = !isWishlisted;
    setIsWishlisted(newWishlistStatus);
    dispatch(toggleWishlistItem(true));

    try {
      if (newWishlistStatus) {
        const result = await dispatch(
          addToWishlist({ user_id, product_id, varient_id: variant.variant_id })
        ).unwrap();
        setWishlistId(result.wishlist_id);
        product.variants[selectedVariantIndex].is_wishlisted = true;
        product.variants[selectedVariantIndex].wishlist_id = result.wishlist_id;
      } else {
        if (!wishlistId) throw new Error('Wishlist ID is missing');
        await dispatch(removeFromWishlist({ wishlist_id: wishlistId })).unwrap();
        setWishlistId(null);
        product.variants[selectedVariantIndex].is_wishlisted = false;
        product.variants[selectedVariantIndex].wishlist_id = null;
      }
    } catch (error) {
      setIsWishlisted(!newWishlistStatus);
      console.error('Wishlist error:', error);
      const errorMessage = typeof error === 'string' ? error : error.message || JSON.stringify(error);
      Alert.alert('Error', errorMessage || 'Wishlist action failed');
    } finally {
      dispatch(toggleWishlistItem(false));
    }
  };

  const handleRecommendedWishlistToggle = async (item) => {
    const productId = item.product_id;
    if (recommendedWishlistLoading[productId]) return;

    // Set loading state
    setRecommendedWishlistLoading((prev) => ({ ...prev, [productId]: true }));
    const newWishlistStatus = !item.wishlist_flag;

    // Optimistic update
    setLocalRecommendedProducts((prev) =>
      prev.map((prod) =>
        prod.product_id === productId
          ? { ...prod, wishlist_flag: newWishlistStatus, wishlist_id: newWishlistStatus ? prod.wishlist_id : null }
          : prod
      )
    );

    try {
      if (newWishlistStatus) {
        // Add to wishlist
        if (!item.varient_id && !item.variant_id) {
          throw new Error('Variant ID is missing');
        }
        const result = await dispatch(
          addToWishlist({
            user_id,
            product_id: productId,
            varient_id: item.varient_id || item.variant_id,
          })
        ).unwrap();
        // Update local state with wishlist_id
        setLocalRecommendedProducts((prev) =>
          prev.map((prod) =>
            prod.product_id === productId
              ? { ...prod, wishlist_flag: true, wishlist_id: result.wishlist_id }
              : prod
          )
        );
        // Optionally refetch to ensure consistency
        dispatch(fetchRecommendedProducts({ user_id, subcategory_id: product.subcategory_id }));
      } else {
        // Remove from wishlist
        if (!item.wishlist_id) throw new Error('Wishlist ID is missing');
        await dispatch(removeFromWishlist({ wishlist_id: item.wishlist_id })).unwrap();
        // Update local state
        setLocalRecommendedProducts((prev) =>
          prev.map((prod) =>
            prod.product_id === productId
              ? { ...prod, wishlist_flag: false, wishlist_id: null }
              : prod
          )
        );
        // Optionally refetch to ensure consistency
        dispatch(fetchRecommendedProducts({ user_id, subcategory_id: product.subcategory_id }));
      }
    } catch (error) {
      // Revert optimistic update
      setLocalRecommendedProducts((prev) =>
        prev.map((prod) =>
          prod.product_id === productId
            ? { ...prod, wishlist_flag: !newWishlistStatus, wishlist_id: item.wishlist_id }
            : prod
        )
      );
      console.error('Recommended Wishlist error:', error);
      const errorMessage = typeof error === 'string' ? error : error.message || JSON.stringify(error);
      Alert.alert('Error', errorMessage || 'Wishlist action failed');
    } finally {
      setRecommendedWishlistLoading((prev) => ({ ...prev, [productId]: false }));
    }
  };

  const handleAddToBag = () => {
    navigation.navigate('Cart');
  };

  if (loading || !product || !product.variants) {
    return (
      <SafeAreaView style={styles.safeArea}>
        <StatusBar translucent backgroundColor="transparent" barStyle="dark-content" />
        <View style={styles.loaderContainer}>
          <ActivityIndicator size="large" color="#000" />
          <Text>Loading product details...</Text>
        </View>
      </SafeAreaView>
    );
  }

  const selectedVariant = product.variants[selectedVariantIndex];
  const selectedSize = selectedVariant.sizes.find((s) => s.size_id === selectedSizeId);
  const salePrice = selectedSize
    ? parseInt(selectedSize.size_price)
    : parseInt(product.variants[0].sizes[0].size_price);
  const mrp = salePrice + 200;

  return (
    <SafeAreaView style={styles.safeArea}>
      <StatusBar translucent backgroundColor="transparent" barStyle="dark-content" />
      <ScrollView
        style={styles.container}
        contentContainerStyle={{
          paddingBottom: insets.bottom + responsiveHeight(8),
        }}
      >
        {/* Top Image */}
        <View style={{ position: 'relative' }}>
          <Image
            source={{ uri: selectedVariant.variant_image }}
            style={{ width: '100%', height: 450, resizeMode: 'cover' }}
          />
          <View style={styles.topOverlay}>
            <TouchableOpacity onPress={() => navigation.goBack()} style={styles.iconButton}>
              <Ionicons name="arrow-back" size={20} color="#000" />
            </TouchableOpacity>
            <TouchableOpacity onPress={toggleWishlist} style={styles.iconButton} disabled={!!wishlistLoading}>
              {wishlistLoading ? (
                <ActivityIndicator size="small" color="#000" />
              ) : (
                <Ionicons
                  name={isWishlisted ? 'heart' : 'heart-outline'}
                  size={20}
                  color={isWishlisted ? 'red' : '#000'}
                />
              )}
            </TouchableOpacity>
          </View>
        </View>

        {/* Product Info */}
        <View style={{ padding: 10, backgroundColor: '#fff' }}>
          <Text style={{ fontSize: 16, fontWeight: 'bold', color: '#000' }}>{product.brand_name}</Text>
          <Text style={{ fontSize: 15, color: '#333', marginVertical: 4 }}>{product.product_name}</Text>
          <View style={{ flexDirection: 'row', alignItems: 'center', gap: 10 }}>
            <Text style={{ fontSize: 16, fontWeight: 'bold', color: '#000' }}>₹{salePrice}</Text>
            <Text style={{ fontSize: 14, textDecorationLine: 'line-through', color: '#999' }}>
              ₹{mrp}
            </Text>
          </View>
        </View>

        {/* Variant Images */}
        <ScrollView
          horizontal
          showsHorizontalScrollIndicator={false}
          contentContainerStyle={styles.imageScrollContainer}
        >
          {product.variants.map((variant, index) => (
            <TouchableOpacity
              key={variant.variant_id}
              onPress={() => handleVariantChange(index)}
            >
              <Image
                source={{ uri: variant.variant_image }}
                style={[
                  styles.productImage,
                  selectedVariantIndex === index && styles.selectedVariantImage,
                ]}
                resizeMode="contain"
              />
            </TouchableOpacity>
          ))}
        </ScrollView>

        {/* Sizes Section */}
        <View style={styles.infoSection}>
          <View style={styles.sizeHeader}>

            </Text>

              <View key={size.size_id} style={{ alignItems: 'center' }}>
                <TouchableOpacity
                  style={[
                    styles.sizeBox,

                  ]}
                  onPress={() => setSelectedSizeId(size.size_id)}

                      selectedSizeId === size.size_id && styles.selectedSizeText,
                    ]}
                  >
                    {size.size_name}
                  </Text>
                </TouchableOpacity>
                {size.size_name === 'L' && <Text style={styles.lowStockText}>2 left</Text>}
              </View>

        </View>

        {/* Pincode Check Section */}
        <View style={{ padding: 16, backgroundColor: '#fff' }}>
          <View style={[commonstyles.row, { gap: 5, justifyContent: 'flex-start' }]}>

    zIndex: 2,

  iconButton: {
    backgroundColor: '#fff',

    elevation: 4,

    borderTopColor: '#333',
    alignItems: 'center',

//cart code 

import React, { useState, useEffect } from 'react';
import {
  View,
  Text,
  Image,
  ScrollView,
  TouchableOpacity,
  StyleSheet,
  Dimensions,
  SafeAreaView,
  StatusBar,
  ActivityIndicator,
  Alert,
} from 'react-native';
import Ionicons from 'react-native-vector-icons/Ionicons';
import { useSafeAreaInsets } from 'react-native-safe-area-context';
import {
  responsiveWidth,
  responsiveHeight,
  responsiveFontSize,
} from 'react-native-responsive-dimensions';
import commonstyles from '../commonstyles/commonstyles';
import { TextInput } from 'react-native-gesture-handler';
import ProductCard from '../components/ProductCard';
import { useDispatch, useSelector } from 'react-redux';
import {
  fetchProductDetails,
  fetchRecommendedProducts,
  fetchSubcategories,
} from '../redux/slices/categorySlice';
import { addToWishlist, removeFromWishlist, toggleWishlistItem } from '../redux/slices/wishlistSlice';
import { addToCart, clearError, resetLastAddedItem } from '../redux/slices/cartSlice';
import { useRoute } from '@react-navigation/native';

const { width } = Dimensions.get('window');

const ProductDetailsScreen = ({ navigation }) => {
  const route = useRoute();
  const dispatch = useDispatch();
  const { user_id, product_id } = route.params;
  const product = useSelector((state) => state.category.productDetails);
  const recommendedProducts = useSelector((state) => state.category.recommendedProducts);
  const subcategories = useSelector((state) => state.category.subcategories);
  const loading = useSelector((state) => state.category.productDetailsLoading);
  const recommendedLoading = useSelector((state) => state.category.recommendedProductsLoading);
  const subLoading = useSelector((state) => state.category.subLoading);
  const subError = useSelector((state) => state.category.subError);
  const wishlistLoading = useSelector((state) => state.wishlist.loading);
  const wishlistError = useSelector((state) => state.wishlist.error);
  const cartLoading = useSelector((state) => state.cart?.loading || false);
  const cartError = useSelector((state) => state.cart?.error || null);
  // Move useSelector to top level for debugging
  const reduxState = useSelector((state) => state);
  const [selectedVariantIndex, setSelectedVariantIndex] = useState(0);
  const [selectedSizeId, setSelectedSizeId] = useState(null);
  const [isWishlisted, setIsWishlisted] = useState(false);
  const [wishlistId, setWishlistId] = useState(null);
  const [localRecommendedProducts, setLocalRecommendedProducts] = useState([]);
  const [recommendedWishlistLoading, setRecommendedWishlistLoading] = useState({});

  const insets = useSafeAreaInsets();

  // Log Redux state for debugging
  useEffect(() => {
    console.log('Redux State:', reduxState);
  }, [reduxState]);

  // Sync localRecommendedProducts with Redux recommendedProducts
  useEffect(() => {
    setLocalRecommendedProducts(recommendedProducts);
  }, [recommendedProducts]);

  useEffect(() => {
    dispatch(fetchProductDetails({ user_id, product_id }));
  }, [user_id, product_id, dispatch]);

  useEffect(() => {
    if (product?.subcategory_id) {
      dispatch(fetchRecommendedProducts({ user_id, subcategory_id: product.subcategory_id, product_id }));
    } else if (product?.category_id) {
      dispatch(fetchSubcategories(product.category_id));
    }
  }, [product, user_id, dispatch]);

  useEffect(() => {
    if (subcategories.length > 0 && !product?.subcategory_id && product?.subcategory_name) {
      const subcategory = subcategories.find(
        (sub) => sub.sub_category_name === product.subcategory_name
      );
      if (subcategory?.id) {
        dispatch(fetchRecommendedProducts({ user_id, subcategory_id: subcategory.id }));
      }
    }
  }, [subcategories, product, user_id, dispatch]);

  useEffect(() => {
    if (product?.variants?.length > 0) {
      const defaultSizeId = product.variants[0].sizes?.[0]?.size_id;
      setSelectedSizeId(defaultSizeId);
      setIsWishlisted(product.variants[0].is_wishlisted || false);
      setWishlistId(product.variants[0].wishlist_id || null);
    }
  }, [product]);

  // Handle cart error
  useEffect(() => {
    if (cartError) {
      Alert.alert('Error', cartError || 'Failed to add to cart');
      dispatch(clearError());
    }
  }, [cartError, dispatch]);

  const handleVariantChange = (index) => {
    setSelectedVariantIndex(index);
    const variant = product.variants[index];
    setSelectedSizeId(variant.sizes[0]?.size_id || null);
    setIsWishlisted(variant.is_wishlisted || false);
    setWishlistId(variant.wishlist_id || null);
  };

  const toggleWishlist = async () => {
    if (wishlistLoading) return;
    const variant = product.variants[selectedVariantIndex];
    const newWishlistStatus = !isWishlisted;
    setIsWishlisted(newWishlistStatus);
    dispatch(toggleWishlistItem(true));

    try {
      if (newWishlistStatus) {
        const result = await dispatch(
          addToWishlist({ user_id, product_id, varient_id: variant.variant_id })
        ).unwrap();
        setWishlistId(result.wishlist_id);
        product.variants[selectedVariantIndex].is_wishlisted = true;
        product.variants[selectedVariantIndex].wishlist_id = result.wishlist_id;
      } else {
        if (!wishlistId) throw new Error('Wishlist ID is missing');
        await dispatch(removeFromWishlist({ wishlist_id: wishlistId })).unwrap();
        setWishlistId(null);
        product.variants[selectedVariantIndex].is_wishlisted = false;
        product.variants[selectedVariantIndex].wishlist_id = null;
      }
    } catch (error) {
      setIsWishlisted(!newWishlistStatus);
      console.error('Wishlist error:', error);
      const errorMessage = typeof error === 'string' ? error : error.message || JSON.stringify(error);
      Alert.alert('Error', errorMessage || 'Wishlist action failed');
    } finally {
      dispatch(toggleWishlistItem(false));
    }
  };

  const handleRecommendedWishlistToggle = async (item) => {
    const productId = item.product_id;
    if (recommendedWishlistLoading[productId]) return;

    setRecommendedWishlistLoading((prev) => ({ ...prev, [productId]: true }));
    const newWishlistStatus = !item.wishlist_flag;

    setLocalRecommendedProducts((prev) =>
      prev.map((prod) =>
        prod.product_id === productId
          ? { ...prod, wishlist_flag: newWishlistStatus, wishlist_id: newWishlistStatus ? prod.wishlist_id : null }
          : prod
      )
    );

    try {
      if (newWishlistStatus) {
        if (!item.varient_id && !item.variant_id) {
          throw new Error('Variant ID is missing');
        }
        const result = await dispatch(
          addToWishlist({
            user_id,
            product_id: productId,
            varient_id: item.varient_id || item.variant_id,
          })
        ).unwrap();
        setLocalRecommendedProducts((prev) =>
          prev.map((prod) =>
            prod.product_id === productId
              ? { ...prod, wishlist_flag: true, wishlist_id: result.wishlist_id }
              : prod
          )
        );
        dispatch(fetchRecommendedProducts({ user_id, subcategory_id: product.subcategory_id }));
      } else {
        if (!item.wishlist_id) throw new Error('Wishlist ID is missing');
        await dispatch(removeFromWishlist({ wishlist_id: item.wishlist_id })).unwrap();
        setLocalRecommendedProducts((prev) =>
          prev.map((prod) =>
            prod.product_id === productId
              ? { ...prod, wishlist_flag: false, wishlist_id: null }
              : prod
          )
        );
        dispatch(fetchRecommendedProducts({ user_id, subcategory_id: product.subcategory_id }));
      }
    } catch (error) {
      setLocalRecommendedProducts((prev) =>
        prev.map((prod) =>
          prod.product_id === productId
            ? { ...prod, wishlist_flag: !newWishlistStatus, wishlist_id: item.wishlist_id }
            : prod
        )
      );
      console.error('Recommended Wishlist error:', error);
      const errorMessage = typeof error === 'string' ? error : error.message || JSON.stringify(error);
      Alert.alert('Error', errorMessage || 'Wishlist action failed');
    } finally {
      setRecommendedWishlistLoading((prev) => ({ ...prev, [productId]: false }));
    }
  };

  const handleAddToBag = async () => {
    if (cartLoading) return;

    const selectedVariant = product.variants[selectedVariantIndex];
    if (!selectedSizeId) {
      Alert.alert('Error', 'Please select a size');
      return;
    }

    const payload = {
      user_id: user_id,
      product_id: product.product_id,
      varient_id: selectedVariant.variant_id,
      varient_size_id: selectedSizeId,
      product_name: product.product_name,
      brand_name: product.brand_name,
      product_color: selectedVariant.color_name,
      product_image: selectedVariant.variant_image || null,
    };

    console.log('Add to Cart Payload:', payload);

    try {
      const result = await dispatch(addToCart(payload)).unwrap();
      Alert.alert('Success', 'Item added to cart successfully', [
        {
          text: 'Go to Cart',
          onPress: () => navigation.navigate('Cart'),
        },
        { text: 'Continue Shopping', style: 'cancel' },
      ]);
      dispatch(resetLastAddedItem());
    } catch (error) {
      console.error('Add to cart error:', error);
    }
  };

  if (loading || !product || !product.variants) {
    return (
      <SafeAreaView style={styles.safeArea}>
        <StatusBar backgroundColor="transparent" barStyle="dark-content" />
        <View style={styles.loaderContainer}>
          <ActivityIndicator size="large" color="#000" />
          <Text>Loading product details...</Text>
        </View>
      </SafeAreaView>
    );
  }

  const selectedVariant = product.variants[selectedVariantIndex];
  const selectedSize = selectedVariant.sizes.find((s) => s.size_id === selectedSizeId);
  const salePrice = selectedSize
    ? parseInt(selectedSize.size_price)
    : parseInt(product.variants[0].sizes?.[0]?.size_price);
  const mrp = salePrice + 200;

  return (
    <SafeAreaView style={styles.safeArea}>
      <StatusBar backgroundColor="transparent" />
      <ScrollView
        style={styles.container}
        contentContainerStyle={{
          paddingBottom: insets.bottom + responsiveHeight(10),
        }}
      >
        {/* Top Image */}
        <View style={{ position: 'relative' }}>
          <Image
            source={{ uri: selectedVariant.variant_image }}
            style={{ width: '100%', height: 450, resizeMode: 'contain' }}
          />
          <View style={styles.topOverlay}>
            <TouchableOpacity onPress={() => navigation.goBack()} style={styles.overlayButton}>
              <Ionicons name="arrow-back" size={20} color="black" />
            </TouchableOpacity>
            <TouchableOpacity onPress={toggleWishlist} style={styles.overlayButton} disabled={!!wishlistLoading}>
              {wishlistLoading ? (
                <ActivityIndicator size="small" color="#666" />
              ) : (
                <Ionicons
                  name={isWishlisted ? 'heart' : 'heart-outline'}
                  size={20}
                  color={isWishlisted ? 'red' : 'black'}
                />
              )}
            </TouchableOpacity>
          </View>
        </View>

        {/* Product Info */}
        <View style={{ padding: 10, backgroundColor: '#fff' }}>
          <Text style={{ fontSize: 16, fontWeight: 'bold', color: '#000' }}>{product.brand_name}</Text>
          <Text style={{ fontSize: 15, color: '#333', marginVertical: 4 }}>{product.product_name}</Text>
          <View style={{ flexDirection: 'row', alignItems: 'center', gap: 10 }}>
            <Text style={{ fontSize: 16, fontWeight: 'bold', color: '#000' }}>₹{salePrice}</Text>
            <Text style={{ fontSize: 14, textDecorationLine: 'line-through', color: '#999' }}>
              ₹{mrp}
            </Text>
          </View>
        </View>

        {/* Variant Images */}
        <ScrollView
          horizontal
          showsHorizontalScrollIndicator={false}
          contentContainerStyle={styles.imageScrollContainer}
        >
          {product.variants.map((variant, index) => (
            <TouchableOpacity
              key={variant.variant_id}
              onPress={() => handleVariantChange(index)}
            >
              <Image
                source={{ uri: variant.variant_image }}
                style={[
                  styles.productImage,
                  selectedVariantIndex === index && styles.selectedVariantImage,
                ]}
                resizeMode="contain"
              />
            </TouchableOpacity>
          ))}
        </ScrollView>

        {/* Sizes Section */}
        <View style={styles.infoSection}>
          <View style={styles.sizeHeader}>
            <Text style={styles.sizeHeaderText}>
              Select Size: {selectedSize?.size_name || selectedVariant.sizes[0]?.size_name}
            </Text>
            <Text style={styles.sizeGuide}>SIZE GUIDE</Text>
          </View>
          <View style={styles.sizeRow}>
            {selectedVariant.sizes?.map((size) => (
              <View key={size.size_id} style={{ alignItems: 'center' }}>
                <TouchableOpacity
                  style={[
                    styles.sizeBox,
                    selectedSizeId === size.size_id && styles.selectedSizeBox,
                    size.size_name === 'L' && selectedSizeId !== size.size_id && styles.lowStockBox,
                  ]}
                  onPress={() => setSelectedSizeId(size.size_id)}
                >
                  <Text
                    style={[
                      styles.sizeText,
                      selectedSizeId === size.size_id && styles.selectedSizeText,
                    ]}
                  >
                    {size.size_name}
                  </Text>
                </TouchableOpacity>
                {size.size_name === 'L' && <Text style={styles.lowStockText}>2 left</Text>}
              </View>
            ))}
          </View>
        </View>

        {/* Pincode Check Section */}
        <View style={{ padding: 16, backgroundColor: '#fff' }}>
          <View style={[commonstyles.row, { gap: 5, justifyContent: 'flex-start' }]}>
            <Ionicons name="location-outline" size={24} color="#000" />
            <Text style={{ fontSize: 14, color: '#333' }}>
              Check your pincode before placing the order.
            </Text>
          </View>
          <View
            style={{
              flexDirection: 'row',
              alignItems: 'center',
              marginTop: 8,
              borderBottomWidth: 1,
              borderBottomColor: '#ccc',
              paddingBottom: 4,
            }}
          >
            <TextInput
              style={{ flex: 1, fontSize: 14, color: '#333' }}
              placeholder="Enter Pin code"
              placeholderTextColor="#999"
              keyboardType="numeric"
            />
            <TouchableOpacity
              style={{
                backgroundColor: '#2c2c60',
                paddingHorizontal: 12,
                paddingVertical: 6,
                borderRadius: 4,
              }}
            >
              <Text style={{ color: '#fff', fontSize: 14 }}>Check</Text>
            </TouchableOpacity>
          </View>
        </View>

        {/* Info Cards */}
        <View
          style={{
            flexDirection: 'row',
            justifyContent: 'space-around',
            paddingVertical: 16,
            backgroundColor: '#fff',
          }}
        >
          <View
            style={{
              alignItems: 'center',
              flex: 1,
              borderWidth: 1,
              borderColor: '#ddd',
              marginHorizontal: 4,
              padding: 8,
              borderRadius: 6,
            }}
          >
            <Image source={require('../assets/secure.png')} style={{ width: 30, height: 30 }} />
            <Text style={{ fontWeight: '600', textAlign: 'center' }}>Secure payments</Text>
            <Text style={{ fontSize: 12, textAlign: 'center', color: '#666' }}>
              Your transactions are safe with us
            </Text>
          </View>
          <View
            style={{
              alignItems: 'center',
              flex: 1,
              borderWidth: 1,
              borderColor: '#ddd',
              marginHorizontal: 4,
              padding: 8,
              borderRadius: 6,
            }}
          >
            <Image source={require('../assets/genuine.png')} style={{ width: 30, height: 30 }} />
            <Text style={{ fontWeight: '600', marginTop: 4, textAlign: 'center' }}>
              Genuine Products
            </Text>
            <Text style={{ fontSize: 12, textAlign: 'center', color: '#666' }}>
              100% authentic, quality checked
            </Text>
          </View>
          <View
            style={{
              alignItems: 'center',
              flex: 1,
              borderWidth: 1,
              borderColor: '#ddd',
              marginHorizontal: 4,
              padding: 8,
              borderRadius: 6,
            }}
          >
            <Image source={require('../assets/return.png')} style={{ width: 30, height: 30 }} />
            <Text style={{ fontWeight: '600', marginTop: 4 }}>7 Days Return</Text>
            <Text style={{ fontSize: 12, textAlign: 'center', color: '#666' }}>
              Easy returns within 7 days
            </Text>
          </View>
        </View>

        {/* Product Description */}
        <View style={{ padding: 16, backgroundColor: '#fff' }}>
          <Text style={{ fontWeight: '700', fontSize: 16, color: '#000' }}>
            Product Description:
          </Text>
          <Text style={{ fontSize: 13, color: '#444', marginTop: 4 }}>
            {product.product_description}
          </Text>
        </View>

        {/* Manufactured By */}
        <View style={{ padding: 16, backgroundColor: '#fff' }}>
          <Text style={{ fontWeight: '700', fontSize: 16, color: '#000' }}>
            Manufactured By:
          </Text>
          <Text style={{ fontSize: 13, color: '#444', marginTop: 4 }}>
            {product.manufactured_by}
          </Text>
        </View>

        {/* Recommended Products Horizontal List */}
        <View style={{ paddingVertical: 10, backgroundColor: '#fff' }}>
          <Text
            style={{
              fontWeight: '700',
              fontSize: 16,
              color: '#000',
              paddingHorizontal: 10,
              marginBottom: 8,
            }}
          >
            You may also like
          </Text>
          {recommendedLoading ? (
            <View style={styles.loaderContainer}>
              <ActivityIndicator size="small" color="#000" />
              <Text>Loading recommended products...</Text>
            </View>
          ) : localRecommendedProducts.length > 0 ? (
            <ScrollView
              horizontal
              showsHorizontalScrollIndicator={false}
              contentContainerStyle={{ paddingHorizontal: 10, gap: 8 }}
            >
              {localRecommendedProducts.map((item) => (
                <View style={{ width: responsiveWidth(45) }} key={item.product_id}>
                  <ProductCard
                    image={{ uri: item.image }}
                    title={item.product_name}
                    brand={item.brand_name}
                    salePrice={parseInt(item.product_price)}
                    badge={item.wishlist_flag ? 'Wishlisted' : null}
                    isWishlisted={item.wishlist_flag}
                    isLoading={recommendedWishlistLoading[item.product_id] || false}
                    onPress={() =>
                      navigation.navigate('ProductDetailsScreen', {
                        user_id,
                        product_id: item.product_id,
                        subcategory_id: item.subcategory_id,
                      })
                    }
                    onWishlistPress={() => handleRecommendedWishlistToggle(item)}
                  />
                </View>
              ))}
            </ScrollView>
          ) : (
            <Text style={{ paddingHorizontal: 16, color: '#666' }}>
              No recommended products available.
            </Text>
          )}
        </View>
      </ScrollView>

      {/* Footer */}
      <View style={[styles.footer, { paddingBottom: insets.bottom + responsiveHeight(2) }]}>
        <TouchableOpacity style={styles.button} onPress={handleAddToBag} disabled={cartLoading}>
          {cartLoading ? (
            <ActivityIndicator size="small" color="#fff" />
          ) : (
            <Text style={styles.buttontext}>Add To Bag</Text>
          )}
        </TouchableOpacity>
      </View>
    </SafeAreaView>
  );
};
